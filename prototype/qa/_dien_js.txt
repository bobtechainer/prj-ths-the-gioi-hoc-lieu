    const mx = (x1 + x2) / 2;
    return `M${x1} ${y1}C${mx} ${y1} ${mx} ${y2} ${x2} ${y2}`;
  }

    const s = CIRCUIT.snapshot(dSim);
    $('#dlit').textContent = s.lit + '/6';
    const crates = $('#dcrates');
    if (crates.childElementCount !== 6){ crates.innerHTML = ''; for (let i = 0; i < 6; i++) crates.appendChild(document.createElement('i')); }
    [...crates.children].forEach((c, i) => c.classList.toggle('f', i < s.lit));

    for (const b of ['A','B','C']){
      const y = DY[b];
      seg(dWirePath(DPOS.main.x + 34, DPOS.main.y, DPOS.sw(b).x - 26, y), s.main);

  /* ---------- khay đồ nghề ---------- */
  function dBuildTray(){
    const board = $('#dboard'); board.innerHTML = '';
    for (const id in DTOOLS){
      const c = document.createElement('div');
      c.className = 'chip'; c.dataset.tool = id;
      c.innerHTML = `<svg width="40" height="40" viewBox="0 0 40 40">${dToolGlyph(id, 40)}</svg>` +
        `<b>${DTOOLS[id].ten}</b><small id="dcount-${id}"></small>`;
      c.addEventListener('click', () => dPickTool(id));
      board.appendChild(c);
    }
    dRefreshTray();
  }
  function dRefreshTray(){
    if (!dSim) return;
    const s = CIRCUIT.snapshot(dSim);
    const cnt = { but:'đo bao nhiêu cũng được', fuse:'còn ' + s.spares.fuse + ' cái', wire:'còn ' + s.spares.wire + ' cuộn' };
    document.querySelectorAll('#dboard .chip').forEach(c => {
      const id = c.dataset.tool;
      const el = c.querySelector('#dcount-' + id); if (el) el.textContent = cnt[id];
      c.classList.toggle('sel', dCarry === id);
      c.classList.toggle('used', (id === 'fuse' && !s.spares.fuse) || (id === 'wire' && !s.spares.wire));
    });
  }
  function dShowInfo(id){
    const o = DTOOLS[id];
    $('#dic-nm').textContent = o.ten;
    $('#dic-nm').style.color = '#2b62a8';
    $('#dic-what').textContent = o.lam;
    const w = $('#dic-warn'); w.textContent = 'Mẹo: ' + o.chiHoi; w.style.display = '';
    $('#dinfocard').classList.add('on');
  }
  const dHideInfo = () => $('#dinfocard').classList.remove('on');

  /* ---------- Mầm đi lại trong phố ---------- */
  function dWalkTo(x, y, onArrive){
    dHero.tx = Math.min(1200, Math.max(160, x));
    dHero.ty = Math.min(590, Math.max(120, y));
    dHero.onArrive = onArrive || null;
    if (dHero.tx !== dHero.x) dHero.face = dHero.tx > dHero.x ? 1 : -1;
  }
  function dTick(dt){
    const dx = dHero.tx - dHero.x, dy = dHero.ty - dHero.y;
    const d = Math.hypot(dx, dy);
    if (d > 2){
      const step = Math.min(d, WALK_SPEED * dt);
      dHero.x += dx / d * step; dHero.y += dy / d * step;
    } else if (dHero.onArrive){ const f = dHero.onArrive; dHero.onArrive = null; f(); }
    dDrawHero(d > 2);
  }
  function dDrawHero(moving){
    if (!dHeroLayer || !dHeroLayer.isConnected){
      const svg = $('#dstage'); if (!svg) return;
      dHeroLayer = E('g', { id:'dhero' }, svg);
    }
    const bob = moving ? Math.sin(performance.now() / 70) * 4 : 0;
    const carry = dCarry
      ? `<g transform="translate(-20,-44)">${dToolGlyph(dCarry, 40)}</g>`
      : '';
    dHeroLayer.innerHTML =
      `<g transform="translate(${dHero.x},${dHero.y - 34 + bob})">
         <ellipse cx="0" cy="36" rx="17" ry="5" fill="rgba(58,46,36,.2)"/>
         <g transform="scale(${1.35 * dHero.face},1.35)"><g transform="translate(-23,-26)">${mamGlyph(mamStage, dCarry ? 'oh' : 'calm')}</g></g>
         ${carry}
       </g>`;
  }
  function dStagePoint(e){
    const svg = $('#dstage');
    return new DOMPoint(e.clientX, e.clientY).matrixTransform(svg.getScreenCTM().inverse());
  }

  /* ---------- thao tác ---------- */
  function dPickTool(id){
    const chip = document.querySelector(`#dboard .chip[data-tool="${id}"]`);
    let x = 640;
    if (chip){
      const r = chip.getBoundingClientRect();
      const svg = $('#dstage');
      const p = new DOMPoint(r.x + r.width / 2, 0).matrixTransform(svg.getScreenCTM().inverse());
      x = Math.min(1150, Math.max(180, p.x));
    }
    dWalkTo(x, 585, () => {
      dCarry = id; dLastAction = performance.now();
      document.body.classList.toggle('probing', id === 'but');
      dShowInfo(id); dRefreshTray(); dDrawHero(false);
      if (id === 'but') dMamSay('Cầm bút rồi! Mấy chấm xanh trên dây là chỗ đo được đó hả bạn?', 'oh', 'but1');
    });
  }
  function dHandleEvents(){
    for (const ev of CIRCUIT.drainEvents(dSim)){
      switch (ev.type){
        case 'zap': {
          dFxZap(); dHero.zapT = 1;
          dSpeak('Ấy chết! Còn điện mà đã đụng vào — may là giật nhẹ. NGẮT CẦU DAO nhánh đó trước đã cậu ơi.');
          dMamSay('Tê hết cả tay… ra là phải ngắt điện rồi mới được thay đồ hả?', 'oh');
          break;
        }
        case 'swap':
          if (ev.fixed){ dMamSay('Sáng rồi sáng rồi! Vậy bệnh nằm đúng chỗ đó thật hả?', 'yay'); }
          else dMamSay('Thay xong mà vẫn tối… vậy chắc bệnh không nằm ở đó nhỉ? Đo tiếp không?', 'oh');
          break;
        case 'no-spare':
          dSpeak('Hết đồ dự phòng rồi. Bấm "Ca mới" làm lại từ đầu, lần này đo kỹ trước rồi hẵng thay.');
          break;
        case 'probe':
          if (!ev.on) dMamSay('Ủa chỗ này mất điện nè. Vậy bệnh nằm trước hay sau chỗ này ta?', 'oh', 'probe-off');
          break;
        case 'main':
          if (!ev.on) dMamSay('Cả phố tối thui luôn! Cầu dao tổng mà ngắt là cả khu mất điện thật hả?', 'oh', 'main-off');
          break;
      }
    }
    const s = CIRCUIT.snapshot(dSim);
    if (s.win && !dDone){ dDone = true; setTimeout(dShowDone, 900); }
    dRender();
  }
  function dFxZap(){
    const g = E('g', {}, dLayerFx);
    g.innerHTML = `<path d="M0 -26l-9 14h11l-9 16" stroke="#f2b722" stroke-width="5" fill="none" stroke-linecap="round"/>`;
    g.setAttribute('transform', `translate(${dHero.x},${dHero.y - 40})`);
    if (dHeroLayer) { dHeroLayer.classList.remove('zapping'); void dHeroLayer.getBBox; dHeroLayer.classList.add('zapping'); }
    setTimeout(() => g.remove(), 500);
  }

  function dShowDone(){
    const s = CIRCUIT.snapshot(dSim);
    $('#th2Probe').textContent = s.probes;
    $('#th2Swap').textContent = s.swaps;
    if (mamStage < 5) mamStage += 1;
    $('#th2Stage').textContent = MAM_STAGE_NAME[mamStage];
    $('#dmamFaceTheory').innerHTML = mamGlyph(mamStage, 'yay');
    $('#dmamGrowIcon').innerHTML = mamGlyph(mamStage, 'yay');
    $('#th2Mam').textContent = s.zaps
      ? 'À, hoá ra phải ngắt cầu dao rồi mới được đụng vào dây. Hồi nãy mình quên, tê hết cả tay. Nhớ đời luôn.'
      : 'Mình đo trước rồi mới thay nên không bị giật phát nào. Đo hai đầu là ra ngay chỗ đứt!';
    dSpeak('Cả phố sáng rồi, cảm ơn cậu. Đọc cái bảng bên phải đi — nghề điện nằm gọn trong đó đấy.');
    dMamSay('Sáng hết 6 nhà rồi! Bạn đọc bảng đi, mình ngồi ngắm đèn.', 'yay');
    $('#scr-dien').classList.add('done-mode');
    $('#theory2').scrollTop = 0;
  }

  function dReset(){
    dSim = CIRCUIT.create(dAttempt++);
    dDone = false; dCarry = null; dSaid.clear();
    for (const k in dProbeMarks) delete dProbeMarks[k];
    document.body.classList.remove('probing');
    $('#scr-dien').classList.remove('done-mode');
    dHero.x = dHero.tx = 640; dHero.y = dHero.ty = 560; dHero.onArrive = null;
    dLastAction = performance.now();
    dBuildScene(); dBuildTray(); dHideInfo();
    dSpeak('Nửa khu phố tối om từ chập tối, bệnh nằm đâu đó trên đường dây. Đồ nghề ở khay dưới — cậu dò giúp tôi.');

  function dCheckStuck(){

    if (performance.now() - dLastAction < 40000) return;

  }

  /* chạm trong phố */
  $('#dstage').addEventListener('click', e => {
    dLastAction = performance.now();
    const sw = e.target.closest('.dswitch');
    if (sw){
      const p = sw.dataset.act === 'main' ? DPOS.main : DPOS.sw(sw.dataset.b);
      const act = sw.dataset.act, b = sw.dataset.b;
      dWalkTo(p.x, p.y + 78, () => {
        act === 'main' ? CIRCUIT.toggleMain(dSim) : CIRCUIT.toggleBranch(dSim, b);
        dHandleEvents();
      });
      return;
    }
    const tp = e.target.closest('.dtp');
    if (tp && dCarry === 'but'){
      const id = tp.dataset.tp, p = DPOS.tp[id];
      dWalkTo(p.x, p.y + 66, () => {
        dProbeMarks[id] = CIRCUIT.probe(dSim, id);
        dHandleEvents();
      });
      return;
    }
    const comp = e.target.closest('.dcomp');
    if (comp){
      const id = comp.dataset.comp;
      const need = id.startsWith('fuse') ? 'fuse' : 'wire';
      const p = id.startsWith('fuse') ? DPOS.fuse(id.slice(-1)) : DPOS.joint(id.slice(-1));
      if (dCarry !== need){
        dMamSay(need === 'fuse' ? 'Chỗ này hình như phải dùng cầu chì mới chứ nhỉ?' : 'Chỗ này chắc cần cuộn dây nối mới đúng không ta?', 'oh');
        return;
      }
      dWalkTo(p.x, p.y + 70, () => {
        const r = CIRCUIT.swap(dSim, id);
        if (r.fixed !== undefined && !r.zap){ dCarry = null; document.body.classList.remove('probing'); dHideInfo(); }
        dHandleEvents(); dDrawHero(false);
      });
      return;
    }
    // chạm đất → đi chơi
    const p = dStagePoint(e);
    if (p.x > 60 && p.x < 1220 && p.y > 80 && p.y < 640) dWalkTo(p.x, p.y);
  });
